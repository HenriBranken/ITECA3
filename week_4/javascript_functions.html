charAt()

The charAt() method returns the character at a specified index in a string. 

            
                let text = 'Hello World';

                char_1 = text.charAt(1);
                char_2 = text.charAt(3);
                char_3 = text.charAt(text.length - 1);

                // What is char_<x>?
char_1 = 'e'
char_2 = 'l'
char_3 = 'd'

indexOf()

indexOf() returns the position of the first occurrence of a value in a string. 

            
                let text = 'Hello world, welcome to the universe.';
                let result = text.indexOf('welcome');

                console.log(result);  // What is `result`?
13

charCodeAt()

The charCodeAt() method returns the Unicode of the character at a specified index in a string. 

            
                let text = 'HELLO WORLD';
                let code = text.charCodeAt(0);

                console.log(code);  // What is `code`?
72

fromCharCode()

The String.fromCharCode() method converts Unicode values to characters. 

            
                let text = String.fromCharCode(72, 69, 76, 76, 79);

                console.log(text);  // What is `text`?
'HELLO'

replace()

The replace() method searches a string for a value (or a regular expression).

The replace() method returns a new string with the value(s) replaced. 

            
                let text = 'Visit Microsoft!';
                let result = text.replace('Microsoft', 'Linux');

                console.log(result);  // What is `result`?
'Visit Linux!'

split()

The split() method splits a string into an array of substrings. 

            
                let text = 'what are we going to eat?';
                // (' ') is used as a separator; the string is split between the words.
                let textArr = text.split(' ');

                console.log(textArr);  // What is `textArr`?
textArr = ['what', 'are', 'we', 'going', 'to', 'eat?']

toUpperCase()

The toUpperCase() method converts the string characters to uppercase letters.

toUpperCase() does not change the original string. 

            
                let text = 'The pyramids of Giza';
                let result = text.toUpperCase();

                console.log(result);  // What is `result`?
result = 'THE PYRAMIDS OF GIZA'

toLowerCase()

The toLowerCase() method converts a string to lowercase letters.

toLowerCase() does not change the original string. 

            
                let text = 'Atlantis, UNDER THE SEA';
                let result = text.toLowerCase();

                console.log(result);
result = 'atlantis, under the sea'

join()

The join() method returns an array as a string. 

            
                let shows = ['Berserk', 'Naruto', 'Bleach', 'PsychoPass'];
                let text = shows.join(' and ');

                console.log(text);  // What is `text`?
result = 'Berserk and Naruto and Bleach and PsychoPass'

pow()

The Math.pow(x, y) returns a value of x raised to the power of y. 

            
                let answer = Math.pow(2, 4);

                console.log(answer);  // What is `answer`?
answer = 16

min()

The Math.min() method returns the number with the smallest value. 

            
                let answer = Math.min(0, 150, 30, -2, 38);

                console.log(answer);  // What is `answer`?
answer = -2

max()

The Math.max() method returns the number with the biggest value. 

            
                let answer = Math.max(0, 150, 30, -2000, 38);

                console.log(answer);  // What is `answer`?
answer = 150

round()

The round() method returns number to the nearest integer. 

            
                let result_1 = Math.round(2.5);
                let result_2 = Math.round(15.2);

                console.log(result_1);
                console.log(result_2);  // What is `result_<x>`?
result_1 = 3
result_2 = 15

User-Defined Function

Functions allow you to create routines that you can use over an over again. You save time in re-using common components, and by designing your own functions, you can extend JavaScript's base language to suit your needs. 

            
                function kelvinToCelsius(kelvin) {
                    let x = kelvin - 273.15;
                    return x;
                }

                let degrees = kelvinToCelsius(84);

                console.log(degrees);  // What is `degrees`?
degrees = -189.15
